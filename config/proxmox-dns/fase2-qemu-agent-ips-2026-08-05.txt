root@proxmox-dns[~]# for id in $(qm list | awk 'NR>1 {print $1}'); do
  name=$(qm config "$id" | awk -F': ' '/^name:/ {print $2}')
  ips=$(qm guest cmd "$id" network-get-interfaces 2>/dev/null |
    awk -F'"' '$2=="ip-address" {print $4}' |
    grep -Ev '^(127\.|::1$|fe80:)' |
    paste -sd ',' -)
  printf "%-6s %-25s %s\n" "$id" "$name" "${ips:-AGENT_SEM_RESPOSTA}"
done
