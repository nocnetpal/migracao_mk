## Confirmacao da eno2

root@px-hubsoft:~# ip -br link show dev eno1
eno1             UP             84:2b:2b:5a:45:55 <BROADCAST,MULTICAST,UP,LOWER_UP>

root@px-hubsoft:~# ip -br link show dev eno2
eno2             DOWN           84:2b:2b:5a:45:57 <BROADCAST,MULTICAST>

root@px-hubsoft:~# ip -d link show dev eno2
3: eno2: <BROADCAST,MULTICAST> mtu 1500 qdisc noop state DOWN mode DEFAULT group default qlen 1000
    link/ether 84:2b:2b:5a:45:57 brd ff:ff:ff:ff:ff:ff promiscuity 0 minmtu 46 maxmtu 9000 addrgenmode eui64 numtxqueues 8 numrxqueues 8 gso_max_size 65536 gso_max_segs 65535
    altname enp1s0f1

root@px-hubsoft:~# cat /sys/class/net/eno2/carrier
cat: /sys/class/net/eno2/carrier: Invalid argument
root@px-hubsoft:~# cat /sys/class/net/eno2/operstate
down
root@px-hubsoft:~# readlink /sys/class/net/eno2/master 2>/dev/null || echo "eno2 sem bridge/master"
eno2 sem bridge/master

# Conclusao: eno2 livre, sem master e administrativamente DOWN. O erro ao ler
# carrier e esperado enquanto a interface estiver administrativamente down.

## Link fisico levantado

root@px-hubsoft:~# ip link set dev eno2 up
root@px-hubsoft:~# ip -br link show dev eno2
eno2             UP             84:2b:2b:5a:45:57 <BROADCAST,MULTICAST,UP,LOWER_UP>
root@px-hubsoft:~# cat /sys/class/net/eno2/carrier
1
root@px-hubsoft:~# cat /sys/class/net/eno2/operstate
up
root@px-hubsoft:~# cat /sys/class/net/eno2/speed
1000
root@px-hubsoft:~# cat /sys/class/net/eno2/duplex
full

# Conclusao: segundo cabo validado, link 1 Gbit/s full duplex na eno2.
